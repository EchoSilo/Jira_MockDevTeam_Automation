"""Pathfinding adapter for workflow recalculation on status divergence."""

import logging
from typing import Optional
import pendulum

from src.orchestrator.pathfinder import WorkflowPathfinder
from src.scheduling.scheduler import Scheduler
from src.scheduling.models import ScheduledAction, ActionStatus
from src.reconciliation.adapters import AdaptationStrategy
from src.reconciliation.reconciler import ReconciliationResult

logger = logging.getLogger(__name__)


class PathfindingAdapter:
    """Handles workflow recalculation when status diverges from expected.

    When reconciliation returns RECALCULATE strategy, this adapter:
    1. Uses WorkflowPathfinder to compute remaining transitions
    2. Marks original scheduled actions as ADAPTED
    3. Inserts new actions for the recalculated path
    """

    def __init__(
        self,
        scheduler: Scheduler,
        pathfinder: Optional[WorkflowPathfinder] = None,
        agent_registry: Optional[dict] = None,
    ):
        """Initialize adapter.

        Args:
            scheduler: The scheduler for action management
            pathfinder: WorkflowPathfinder instance (creates default if None)
            agent_registry: Mapping of role -> available agent_ids
        """
        self.scheduler = scheduler
        self.pathfinder = pathfinder or WorkflowPathfinder()
        self.agent_registry = agent_registry or {}

    def handle_recalculate(
        self,
        ticket_key: str,
        actual_status: str,
        target_status: str = "Done",
        scenario_id: Optional[str] = None,
        original_action_id: Optional[str] = None,
    ) -> dict:
        """Handle RECALCULATE strategy by computing and scheduling new path.

        Args:
            ticket_key: The Jira ticket key
            actual_status: Current actual status in Jira
            target_status: Target status to reach (default "Done")
            scenario_id: Scenario ID for the new actions
            original_action_id: ID of the original action being adapted

        Returns:
            Dict with recalculation results:
            - adapted_actions: list of action IDs marked as ADAPTED
            - new_actions: list of new action IDs scheduled
            - path: list of computed path steps
        """
        logger.info(f"Recalculating path for {ticket_key} from '{actual_status}' to '{target_status}'")

        # Compute new path using BFS
        path = self.pathfinder.recalculate_remaining_script(
            current_status=actual_status,
            target_status=target_status.lower(),
        )

        if not path:
            logger.warning(f"No path found from '{actual_status}' to '{target_status}'")
            return {
                "adapted_actions": [],
                "new_actions": [],
                "path": [],
            }

        adapted = []
        new_actions = []

        # Mark original action as adapted
        if original_action_id:
            self._mark_action_adapted(
                original_action_id,
                f"Status diverged to '{actual_status}', recalculating path"
            )
            adapted.append(original_action_id)

        # Mark any other pending actions for this ticket as adapted
        pending = self._get_pending_actions_for_ticket(ticket_key)
        for action in pending:
            if action.action_id != original_action_id:
                self._mark_action_adapted(
                    action.action_id,
                    f"Path recalculated from '{actual_status}'"
                )
                adapted.append(action.action_id)

        # Schedule new actions for the computed path
        base_time = pendulum.now("UTC")
        for i, step in enumerate(path):
            agent_id = self._select_agent_for_role(step.get("role", "developer"))
            if not agent_id:
                logger.warning(f"No agent available for role {step.get('role')}")
                continue

            new_action = ScheduledAction(
                scheduled_time=base_time.add(hours=i + 1),
                action_type=step["type"],
                agent_id=agent_id,
                ticket_key=ticket_key,
                scenario_id=scenario_id,
                expected_status=step.get("target_status"),
                params={
                    "adapted": True,
                    "reason": "status_divergence",
                    "original_status": actual_status,
                },
            )
            self.scheduler.schedule_action(new_action)
            new_actions.append(new_action.action_id)

        logger.info(
            f"Recalculated {ticket_key}: adapted {len(adapted)}, new {len(new_actions)} actions"
        )

        return {
            "adapted_actions": adapted,
            "new_actions": new_actions,
            "path": path,
        }

    def handle_reconciliation_result(
        self,
        result: ReconciliationResult,
        action: ScheduledAction,
    ) -> Optional[dict]:
        """Handle reconciliation result, processing RECALCULATE if needed.

        Args:
            result: The reconciliation result from ReconciliationEngine
            action: The ScheduledAction being reconciled

        Returns:
            Recalculation result dict if RECALCULATE, None otherwise
        """
        if result.strategy != AdaptationStrategy.RECALCULATE:
            return None

        # Extract actual status from reason (format: "Ticket regressed to 'X'...")
        actual_status = self._extract_status_from_reason(result.reason)
        if not actual_status:
            # Fallback: use action's expected status
            actual_status = action.expected_status or "To Do"

        return self.handle_recalculate(
            ticket_key=action.ticket_key,
            actual_status=actual_status,
            scenario_id=action.scenario_id,
            original_action_id=action.action_id,
        )

    def _extract_status_from_reason(self, reason: str) -> Optional[str]:
        """Extract status from reconciliation reason string."""
        # Pattern: "Ticket regressed to 'In Progress'..."
        import re
        match = re.search(r"to '([^']+)'", reason)
        if match:
            return match.group(1)
        return None

    def _get_pending_actions_for_ticket(self, ticket_key: str) -> list[ScheduledAction]:
        """Get pending actions for a specific ticket."""
        return [
            a for a in self.scheduler.queue._heap
            if a.ticket_key == ticket_key and a.status == ActionStatus.PENDING
        ]

    def _mark_action_adapted(self, action_id: str, reason: str) -> None:
        """Mark action as adapted with reason."""
        for action in self.scheduler.queue._heap:
            if action.action_id == action_id:
                action.status = ActionStatus.ADAPTED
                action.result = {"adapted_reason": reason}
                self.scheduler.store.update_status(
                    action_id,
                    ActionStatus.ADAPTED,
                    result=action.result,
                )
                logger.debug(f"Marked {action_id} as ADAPTED: {reason}")
                break

    def _select_agent_for_role(self, role: str) -> Optional[str]:
        """Select an available agent for the given role."""
        agents = self.agent_registry.get(role, [])
        if agents:
            return agents[0]
        return None

import pytest
import pendulum
from unittest.mock import MagicMock, patch

from src.chaos.pathfinding_adapter import PathfindingAdapter
from src.scheduling.models import ScheduledAction, ActionStatus
from src.scheduling.scheduler import Scheduler
from src.reconciliation.adapters import AdaptationStrategy
from src.reconciliation.reconciler import ReconciliationResult


@pytest.fixture
def mock_scheduler():
    """Create a mock scheduler with an empty queue."""
    scheduler = MagicMock(spec=Scheduler)
    scheduler.queue = MagicMock()
    scheduler.queue._heap = []
    scheduler.store = MagicMock()
    return scheduler


@pytest.fixture
def mock_pathfinder():
    """Create a mock pathfinder."""
    pathfinder = MagicMock()
    # Default: return 2-step path to Done
    pathfinder.recalculate_remaining_script.return_value = [
        {"type": "push_to_review", "target_status": "Code Review", "role": "developer"},
        {"type": "qa_approve", "target_status": "Done", "role": "qa"},
    ]
    return pathfinder


@pytest.fixture
def agent_registry():
    """Sample agent registry."""
    return {
        "developer": ["dev-alice"],
        "qa": ["qa-charlie"],
        "tech_lead": ["tl-diana"],

@pytest.fixture
def adapter(mock_scheduler, mock_pathfinder, agent_registry):
    """Create adapter with mocks."""
    return PathfindingAdapter(mock_scheduler, mock_pathfinder, agent_registry)


class TestPathfindingAdapter:
    """Tests for PathfindingAdapter class."""

    def test_handle_recalculate_schedules_new_actions(
        self, adapter, mock_scheduler, mock_pathfinder
    ):
        """handle_recalculate schedules actions for computed path."""
        result = adapter.handle_recalculate(
            ticket_key="TEST-123",
            actual_status="To Do",
            target_status="Done",
            scenario_id="scenario-1",
        )

        # Should have scheduled 2 actions (one per path step)
        assert mock_scheduler.schedule_action.call_count == 2
        assert len(result["new_actions"]) == 2
        assert len(result["path"]) == 2

    def test_handle_recalculate_marks_original_adapted(
        self, adapter, mock_scheduler, mock_pathfinder
    ):
        """handle_recalculate marks original action as adapted."""
        original_action = ScheduledAction(
            scheduled_time=pendulum.now("UTC"),
            action_type="dev_progress",
            agent_id="dev-alice",
            ticket_key="TEST-123",
        )
        mock_scheduler.queue._heap = [original_action]

        result = adapter.handle_recalculate(
            ticket_key="TEST-123",
            actual_status="To Do",
            original_action_id=original_action.action_id,
        )

        assert original_action.action_id in result["adapted_actions"]
        assert original_action.status == ActionStatus.ADAPTED

    def test_handle_recalculate_marks_pending_adapted(
        self, adapter, mock_scheduler, mock_pathfinder
    ):
        """handle_recalculate marks other pending actions for ticket as adapted."""
        pending_action = ScheduledAction(
            scheduled_time=pendulum.now("UTC").add(hours=2),
            action_type="push_to_qa",
            agent_id="dev-alice",
            ticket_key="TEST-123",
        )
        mock_scheduler.queue._heap = [pending_action]

        result = adapter.handle_recalculate(
            ticket_key="TEST-123",
            actual_status="To Do",
        )

        assert pending_action.action_id in result["adapted_actions"]
        assert pending_action.status == ActionStatus.ADAPTED

    def test_handle_recalculate_no_path(self, adapter, mock_scheduler, mock_pathfinder):
        """handle_recalculate handles no path gracefully."""
        mock_pathfinder.recalculate_remaining_script.return_value = []

        result = adapter.handle_recalculate(
            ticket_key="TEST-123",
            actual_status="Done",  # Already at target
        )

        assert len(result["new_actions"]) == 0
        assert len(result["path"]) == 0

    def test_handle_reconciliation_result_recalculate(
        self, adapter, mock_scheduler, mock_pathfinder
    ):
        """handle_reconciliation_result processes RECALCULATE."""
        result = ReconciliationResult(

    def test_handle_reconciliation_result_non_recalculate(

    ):

    def test_extract_status_from_reason(self, adapter):

        status = adapter._extract_status_from_reason(reason)
