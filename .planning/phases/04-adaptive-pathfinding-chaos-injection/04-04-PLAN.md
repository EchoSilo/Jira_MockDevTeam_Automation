"""Scenario adapter for modifying scenarios in response to chaos events."""

import logging
from typing import Optional
import pendulum

from src.scheduling.scheduler import Scheduler
from src.scheduling.models import ScheduledAction, ActionStatus
from .models import RandomEvent, ChaosEventType
from .event_catalog import EventCatalog

logger = logging.getLogger(__name__)


class AdaptationResult:
    """Result of scenario adaptation."""

    def __init__(
        self,
        event_id: str,
        inserted_actions: list[str],
        adapted_actions: list[str],
        reassigned_actions: dict[str, str] = None,  # old_action_id -> new_action_id
    ):
        self.event_id = event_id
        self.inserted_actions = inserted_actions
        self.adapted_actions = adapted_actions
        self.reassigned_actions = reassigned_actions or {}

    def to_dict(self) -> dict:
        return {
            "event_id": self.event_id,
            "inserted_count": len(self.inserted_actions),
            "adapted_count": len(self.adapted_actions),
            "reassigned_count": len(self.reassigned_actions),
        }


class ScenarioAdapter:
    """Modifies active scenarios when random events occur."""

    def __init__(
        self,
        scheduler: Scheduler,
        event_catalog: Optional[EventCatalog] = None,
        agent_registry: Optional[dict] = None,  # role -> list of agent_ids
    ):
        """Initialize adapter.

        Args:
            scheduler: The scheduler for action management
            event_catalog: Event templates and config
            agent_registry: Mapping of role -> available agent_ids
        """
        self.scheduler = scheduler
        self.catalog = event_catalog or EventCatalog()
        self.agent_registry = agent_registry or {}

    def adapt_to_event(self, event: RandomEvent) -> AdaptationResult:
        """Adapt scheduled actions in response to random event.

        Args:
            event: The chaos event that occurred

        Returns:
            AdaptationResult with inserted/adapted action counts
        """
        logger.info(f"Adapting to chaos event: {event.event_type.value} ({event.event_id})")

        if event.event_type == ChaosEventType.URGENT_BUG:
            return self._handle_urgent_bug(event)
        elif event.event_type == ChaosEventType.PRODUCTION_OUTAGE:
            return self._handle_production_outage(event)
        elif event.event_type == ChaosEventType.TEAM_ABSENCE:
            return self._handle_team_absence(event)
        elif event.event_type == ChaosEventType.EXTERNAL_BLOCKER:
            return self._handle_external_blocker(event)
        elif event.event_type == ChaosEventType.PRIORITY_SHIFT:
            return self._handle_priority_shift(event)
        elif event.event_type == ChaosEventType.SCOPE_CHANGE:
            return self._handle_scope_change(event)
        else:
            logger.warning(f"Unknown event type: {event.event_type}")
            return AdaptationResult(event.event_id, [], [])

    def _handle_urgent_bug(self, event: RandomEvent) -> AdaptationResult:
        """Handle urgent bug: insert fix actions, postpone other work."""
        inserted = []
        adapted = []

        for ticket_key in event.affected_tickets:
            # Postpone existing pending actions for this ticket
            pending = self._get_pending_actions_for_ticket(ticket_key)
            for action in pending:
                self._mark_action_adapted(action, f"Postponed due to urgent bug {event.event_id}")
                adapted.append(action.action_id)

            # Insert urgent fix action
            fix_action = self._create_response_action(
                event=event,
                ticket_key=ticket_key,
                action_type="fix_urgent_bug",
                agent_role="developer",
                delay_hours=1,
            )
            if fix_action:
                self.scheduler.schedule_action(fix_action)
                inserted.append(fix_action.action_id)

        return AdaptationResult(event.event_id, inserted, adapted)

    def _handle_production_outage(self, event: RandomEvent) -> AdaptationResult:
        """Handle production outage: pause non-critical work, insert emergency response."""
        inserted = []
        adapted = []

        # Pause ALL non-critical pending actions
        all_pending = self._get_all_pending_actions()
        for action in all_pending:
            if not self._is_critical_action(action):
                self._mark_action_adapted(action, f"Paused during production outage {event.event_id}")
                adapted.append(action.action_id)

        # Insert emergency response for affected tickets
        for ticket_key in event.affected_tickets:
            response_action = self._create_response_action(
                event=event,
                ticket_key=ticket_key,
                action_type="emergency_response",
                agent_role="tech_lead",
                delay_hours=0.25,  # 15 minutes
            )
            if response_action:
                self.scheduler.schedule_action(response_action)
                inserted.append(response_action.action_id)

        return AdaptationResult(event.event_id, inserted, adapted)

    def _handle_team_absence(self, event: RandomEvent) -> AdaptationResult:
        """Handle team member absence: reassign their scheduled actions."""
        inserted = []
        adapted = []
        reassigned = {}

        absent_agent = event.affected_agents[0] if event.affected_agents else None
        if not absent_agent:
            logger.warning("Team absence event has no affected_agents")
            return AdaptationResult(event.event_id, [], [])

        # Find replacement agent with same role
        replacement = self._find_replacement_agent(absent_agent)
        if not replacement:
            logger.warning(f"No replacement found for absent agent {absent_agent}")
            # Just mark as adapted without replacement
            pending = self._get_pending_actions_for_agent(absent_agent)
            for action in pending:
                self._mark_action_adapted(action, f"Agent {absent_agent} absent, no replacement")
                adapted.append(action.action_id)
            return AdaptationResult(event.event_id, [], adapted)

        # Reassign pending actions to replacement
        pending = self._get_pending_actions_for_agent(absent_agent)
        for action in pending:
            # Mark original as adapted
            self._mark_action_adapted(action, f"Reassigned from absent {absent_agent} to {replacement}")
            adapted.append(action.action_id)

            # Create replacement action
            new_action = ScheduledAction(
                scheduled_time=action.scheduled_time,
                action_type=action.action_type,
                agent_id=replacement,
                ticket_key=action.ticket_key,
                scenario_id=action.scenario_id,
                window_minutes=action.window_minutes,
                expected_status=action.expected_status,
                expected_assignee=action.expected_assignee,
                params={**action.params, "reassigned_from": absent_agent},
            )
            self.scheduler.schedule_action(new_action)
            inserted.append(new_action.action_id)
            reassigned[action.action_id] = new_action.action_id

        return AdaptationResult(event.event_id, inserted, adapted, reassigned)

    def _handle_external_blocker(self, event: RandomEvent) -> AdaptationResult:
        """Handle external blocker: add discussion actions, extend timelines."""
        inserted = []
        adapted = []

        for ticket_key in event.affected_tickets:
            # Insert blocker discussion action
            discuss_action = self._create_response_action(
                event=event,
                ticket_key=ticket_key,
                action_type="blocker_discussion",
                agent_role="pm",
                delay_hours=2,
            )
            if discuss_action:
                self.scheduler.schedule_action(discuss_action)
                inserted.append(discuss_action.action_id)

            # Mark pending actions as adapted (timeline extended)
            pending = self._get_pending_actions_for_ticket(ticket_key)
            for action in pending:
                self._mark_action_adapted(action, f"Extended due to blocker {event.event_id}")
                adapted.append(action.action_id)

        return AdaptationResult(event.event_id, inserted, adapted)

    def _handle_priority_shift(self, event: RandomEvent) -> AdaptationResult:
        """Handle priority shift: reorder actions but don't cancel them."""
        # Priority shifts don't require immediate adaptation
        # The scheduler will handle reordering based on new priorities
        logger.info(f"Priority shift for {event.affected_tickets} - no immediate action needed")
        return AdaptationResult(event.event_id, [], [])

    def _handle_scope_change(self, event: RandomEvent) -> AdaptationResult:
        """Handle scope change: extend timeline for affected tickets."""
        adapted = []

        for ticket_key in event.affected_tickets:
            pending = self._get_pending_actions_for_ticket(ticket_key)
            for action in pending:
                self._mark_action_adapted(action, f"Scope changed {event.event_id}")
                adapted.append(action.action_id)

        return AdaptationResult(event.event_id, [], adapted)

    def _get_pending_actions_for_ticket(self, ticket_key: str) -> list[ScheduledAction]:
        """Get pending actions for a specific ticket."""
        pending = []
        for action in self.scheduler.queue._heap:
            if action.ticket_key == ticket_key and action.status == ActionStatus.PENDING:
                pending.append(action)
        return pending

    def _get_pending_actions_for_agent(self, agent_id: str) -> list[ScheduledAction]:
        """Get pending actions for a specific agent."""
        pending = []
        for action in self.scheduler.queue._heap:
            if action.agent_id == agent_id and action.status == ActionStatus.PENDING:
                pending.append(action)
        return pending

    def _get_all_pending_actions(self) -> list[ScheduledAction]:
        """Get all pending actions."""
        return [a for a in self.scheduler.queue._heap if a.status == ActionStatus.PENDING]

    def _mark_action_adapted(self, action: ScheduledAction, reason: str) -> None:
        """Mark action as adapted with reason."""
        action.status = ActionStatus.ADAPTED
        action.result = {"adapted_reason": reason}
        self.scheduler.store.update_status(
            action.action_id,
            ActionStatus.ADAPTED,
            result=action.result,
        )
        logger.debug(f"Marked action {action.action_id} as ADAPTED: {reason}")

    def _is_critical_action(self, action: ScheduledAction) -> bool:
        """Check if action is critical (should not be paused)."""
        critical_types = {"emergency_response", "fix_urgent_bug", "blocker_resolution"}
        return action.action_type in critical_types

    def _create_response_action(
        self,
        event: RandomEvent,
        ticket_key: str,
        action_type: str,
        agent_role: str,
        delay_hours: float = 1,
    ) -> Optional[ScheduledAction]:
        """Create a response action for an event."""
        agent_id = self._select_agent_for_role(agent_role)
        if not agent_id:
            logger.warning(f"No agent available for role {agent_role}")
            return None

        return ScheduledAction(
            scheduled_time=pendulum.now("UTC").add(hours=delay_hours),
            action_type=action_type,
            agent_id=agent_id,
            ticket_key=ticket_key,
            scenario_id=event.event_id,
            params={
                "event_type": event.event_type.value,
                "severity": event.severity,
                "chaos_response": True,
            },
        )

    def _select_agent_for_role(self, role: str) -> Optional[str]:
        """Select an available agent for the given role."""
        agents = self.agent_registry.get(role, [])
        if agents:
            return agents[0]  # Simple selection - first available
        return None

    def _find_replacement_agent(self, absent_agent: str) -> Optional[str]:
        """Find a replacement agent with the same role."""
        # Find which role the absent agent has
        for role, agents in self.agent_registry.items():
            if absent_agent in agents:
                # Find another agent in the same role
                for agent in agents:
                    if agent != absent_agent:
                        return agent
        return None
