  <behavior>
    AdaptationStrategy enum:
    - CANCEL: Cancel action, mark scenario as invalidated
    - RECALCULATE: Recompute action plan from current Jira state
    - RESCHEDULE: Retry action later (transient failure)
    - PROCEED: Minor divergence, safe to proceed
    - SKIP: Skip this action but continue scenario

    ReconciliationResult dataclass:
    - strategy: AdaptationStrategy
    - reason: str
    - tombstone_reason: Optional[str]  # Why scenario was invalidated (for CANCEL)
    - new_plan: Optional[list]  # Recalculated action plan (for RECALCULATE)

    reconcile_status_mismatch(ticket_key, expected_status, actual_status, action_type):
    - "In Progress" -> "Done" => CANCEL with tombstone "User completed ticket manually (Done)"
    - "In Progress" -> "Code Review" => SKIP "Ticket already progressed to 'Code Review' externally"
    - "Code Review" -> "In Progress" => RECALCULATE "Ticket regressed to 'In Progress', recalculating plan"
    - Unknown statuses => PROCEED cautiously

    reconcile_sprint_mismatch(ticket_key, expected_in_sprint, actual_in_sprint):
    - expected=True, actual=False => CANCEL with tombstone "Removed from sprint by user"
    - expected=False, actual=True => PROCEED "Ticket added to sprint externally"

    reconcile_api_failure(error, retry_count):
    - "timeout" | "503" | "429" | "connection" errors, retry_count < 3 => RESCHEDULE
    - "404" | "not found" => CANCEL with tombstone "Ticket deleted from Jira"
    - Other errors => SKIP "Jira API error (...), skipping action"
  </behavior>
